function openChapter(chapter) { const subject = selected; const key = subject + "|" + chapter; const storageKey = "SR_CLASS10_QUESTION_BANK"; function getBank() { try { return JSON.parse(localStorage.getItem(storageKey) || "{}"); } catch { return {}; } } function saveBank(bank) { localStorage.setItem(storageKey, JSON.stringify(bank)); } const bank = getBank(); const questions = bank[key] || []; const modal = document.createElement("div"); modal.className = "modal-back open"; modal.innerHTML = `

${chapter}

${subject} • अध्यायवार प्रश्न-उत्तर

+ नया प्रश्न जोड़ें
`; document.body.appendChild(modal); function renderQuestions() { const list = modal.querySelector("#questionList"); const current = getBank()[key] || []; if (current.length === 0) { list.innerHTML = `

अभी इस अध्याय में प्रश्न नहीं हैं। नीचे नया प्रश्न जोड़ें।

`; return; } list.innerHTML = current.map((item, i) => `
प्रश्न ${i + 1} • ${item.type}

${escapeHTML(item.question)}

${escapeHTML(item.answer)}
`).join(""); list.querySelectorAll("[data-answer]").forEach(button => { button.onclick = () => { const box = list.querySelector( '[data-answer-box="' + button.dataset.answer + '"]' ); box.classList.toggle("hidden"); button.textContent = box.classList.contains("hidden") ? "उत्तर देखें" : "उत्तर छिपाएँ"; }; }); } function escapeHTML(text) { return String(text).replace(/[&<>"']/g, char => ({ "&": "&", "<": "<", ">": ">", '"': """, "'": "'" })[char]); } renderQuestions(); modal.querySelector(".close").onclick = () => { modal.remove(); }; modal.addEventListener("click", event => { if (event.target === modal) { modal.remove(); } }); modal.querySelector("#addQuestion").onsubmit = event => { event.preventDefault(); const form = new FormData(event.target); const bank = getBank(); if (!bank[key]) { bank[key] = []; } bank[key].push({ question: form.get("question").trim(), answer: form.get("answer").trim(), type: form.get("type") }); saveBank(bank); event.target.reset(); renderQuestions(); alert("प्रश्न सुरक्षित हो गया!"); }; modal.querySelector("#downloadQuestions").onclick = () => { const data = getBank(); const blob = new Blob( [JSON.stringify(data, null, 2)], { type: "application/json" } ); const url = URL.createObjectURL(blob); const link = document.createElement("a"); link.href = url; link.download = "SR-Class10-Questions.json"; link.click(); URL.revokeObjectURL(url); }; }
